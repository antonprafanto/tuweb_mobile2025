💻 1. Halaman Pengirim (HomePage.vue)

import { useRouter } from 'vue-router';

const router = useRouter();

const bukaDetail = (kode: string) => {
  // Navigasi tumpukan dengan parameter URL
  router.push(`/modul/${kode}`);
};

💻 2. Halaman Penerima (ModulDetailPage.vue)

import { useRoute } from 'vue-router';
import { computed } from 'vue';

const route = useRoute();

// Mengambil parameter :kode dari URL aktif
const kodeModul = computed(() => {
  return route.params.kode as string;
});
💡 Tips Dosen: Gunakan computed saat membaca route.params agar komponen otomatis merespons jika parameter URL berganti tanpa memuat ulang komponen dari awal.
