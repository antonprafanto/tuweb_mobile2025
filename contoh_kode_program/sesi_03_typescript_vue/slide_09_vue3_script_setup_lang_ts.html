Integrasi Vue 3 & TypeScript: <script setup lang="ts">

Standar industri pengembangan aplikasi Ionic Vue modern dengan performa tinggi.

📄 Berkas SFC: ProfilTutor.vue

<script setup lang="ts">
import { ref } from 'vue';

// 1. Impor atau Definisikan Interface
interface Dosen {
  nama: string;
  nidn: string;
  isOnline: boolean;
}

// 2. Typing State Reaktif Eksplisit
const dosen = ref<Dosen>({
  nama: 'Anton Prafanto, S.Kom., M.T.',
  nidn: '0012058401',
  isOnline: true
});

const toggleOnline = (): void => {
  dosen.value.isOnline = !dosen.value.isOnline;
};
</script>

NIDN: {{ dosen.nidn }}
