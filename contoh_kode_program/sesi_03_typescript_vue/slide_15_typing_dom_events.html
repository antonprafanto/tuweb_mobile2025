Penanganan DOM Events Bertipe (Event, KeyboardEvent, HTMLInputElement)

Mengakses properti event.target.value secara aman menggunakan Type Casting.

💡 Solusi Type Casting di TypeScript:

// Tanpa TypeScript casting, e.target.value akan dianggap error!
const handleInput = (event: Event): void => {
  // Lakukan Type Assertion (as HTMLInputElement)
  const target = event.target as HTMLInputElement;
  console.log('Nilai input:', target.value);
};

// Khusus Event Keyboard:
const handleKey = (event: KeyboardEvent): void => {
  if (event.key === 'Enter') {
    console.log('Pengguna menekan tombol Enter!');
  }
};
