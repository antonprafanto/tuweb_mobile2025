💡 Kode Definisi TypeScript:

// 1. Literal Types: Hanya mengizinkan teks yang terdaftar
export type NilaiHuruf = 'A' | 'B' | 'C' | 'D' | 'E';
export type StatusMahasiswa = 'Aktif' | 'Cuti' | 'Lulus';

// 2. Union Types pada properti:
export interface RiwayatNilai {
  kodeMk: string;
  nilai: NilaiHuruf;        // Menolak jika diisi selain A, B, C, D, E
  status: StatusMahasiswa;   // Menolak jika diisi "Pending" atau kata sembarangan
}
