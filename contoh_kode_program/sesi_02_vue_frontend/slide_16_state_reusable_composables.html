💡 Struktur Berkas Composable: useKRSManager.js

// Logika bisnis terisolasi 100% dari elemen DOM/HTML!
export function useKRSManager() {
  const krs = ref([]);
  const totalSks = computed(() => krs.value.reduce((sum, item) => sum + item.sks, 0));
  const isPenuh = computed(() => totalSks.value >= 24);

  const ambilMk = (mk) => {
    if (totalSks.value + mk.sks <= 24) krs.value.push(mk);
  };
  const lepasMk = (id) => {
    krs.value = krs.value.filter(item => item.id !== id);
  };

  return { krs, totalSks, isPenuh, ambilMk, lepasMk };
}
